pre_filtered_stocks = []
                for code in tqdm(stock_list, desc="基本筛选"):
                    #if code == '002365.SZ':
                    #     pass
                    try:
                        # 获取数据进行筛选
                        data = self.data_loader.get_stock_data(code)
                        if data is None or data.empty:
                            continue
                            
                        # 获取分割日期前最后一条记录
                        train_data = data[data['date'] <= self.split_date]
                        if train_data.empty:
                            continue
                            
                        last_record = train_data.iloc[-1]
                        
                        # 价格筛选
                        if not (self.min_price <= last_record['close'] <= self.max_price):#最小价格3元，最大价格18元
                            continue
                        
                        # 市值筛选
                        mcap = None
                        try:
                            from xtquant.xtdata import get_instrument_detail
                            stock_info = get_instrument_detail(code)
                            # 计算市值（亿元）
                            mcap = stock_info['FloatVolume'] * last_record['close'] / 100000000
                        except Exception as e:
                            logger.error(f"获取股票 {code} 市值信息出错: {e}")
                            continue
                        
                        if not (self.min_mcap <= mcap <= self.max_mcap):
                            continue
                        
                        # 添加到预筛选股票池
                        pre_filtered_stocks.append(code)
